local sti=type(setthreadidentity)=="function"
local ex=7
if type(getthreadidentity)=="function" then
 local o,i=pcall(getthreadidentity)
 if o and type(i)=="number" then ex=i end
end
local function gc(f,...)
 if type(f)~="function" then return false end
 if sti then pcall(setthreadidentity,2) end
 local r=table.pack(pcall(f,...))
 if sti then pcall(setthreadidentity,ex) end
 return table.unpack(r,1,r.n)
end

local L
do
 local o,m=gc(function()
  return require(R:WaitForChild("Framework",5):WaitForChild("Library",5))
 end)
 if o then L=m end
end

local function fu(f,n)
 if type(f)~="function" or type(debug.getupvalues)~="function" then return end
 local o,u=pcall(debug.getupvalues,f)
 if not o or type(u)~="table" then return end
 for _,v in pairs(u) do
  if type(v)=="function" then
   local o2,nm=pcall(debug.info,v,"n")
   if o2 and nm==n then return v end
  end
 end
end
local gn,rm
local function rs()
 if gn and rm then return true end
 local rf=fu(L and L.Network and L.Network.Invoke,"_remoteFunction")
 local a,b=fu(rf,"_getName"),fu(rf,"_remote")
 if a and b then gn,rm=a,b return true end
 return false
end
local rc={}
local function rem(k,n)
 local key=k..n
 local r=rc[key]
 if r and r.Parent then return r end
 if not rs() then return end
 local o,i=pcall(function() return rm(k,gn(k,n)) end)
 if o and typeof(i)=="Instance" then rc[key]=i return i end
end
local function inv(n,...)
 local r=rem(2,n)
 if not r then return false end
 return gc(r.InvokeServer,r,...)
end
local function fire(n,...)
 local r=rem(1,n)
 if not r then return false end
 return gc(r.FireServer,r,...)
end
local function on(n,f)
 local r=rem(1,n)
 if r then cn[#cn+1]=r.OnClientEvent:Connect(function(...) pcall(f,...) end) end
end

local function tf(n)
 local t=workspace:FindFirstChild("__THINGS")
 return t and t:FindFirstChild(n)
end
local function hr()
 local c=lp.Character
 return c and c:FindFirstChild("HumanoidRootPart")
end
local function pets()
 local l={}
 if not(L and L.PetCmds) then return l end
 local o,e=gc(L.PetCmds.GetEquipped)
 if o and type(e)=="table" then
  for _,p in ipairs(e) do
   if type(p)=="table" and type(p.uid)=="string" then l[#l+1]=p.uid end
  end
 end
 return l
end
local function ha(a)
 if type(a)~="string" or a=="" then return true end
 if not(L.Directory and L.Directory.Areas and L.Directory.Areas[a]) then return false end
 if not L.WorldCmds then return false end
 local o,r=gc(L.WorldCmds.HasArea,a)
 return o and r==true
end
local function cda()
 if not(L and L.WorldCmds and type(L.WorldCmds.CanDoAction)=="function") then return true end
 local o,r=gc(L.WorldCmds.CanDoAction)
 return not(o and r==false)
end
local function cp(f)
 local p=f:FindFirstChild("POS")
 if p and p:IsA("BasePart") then return p.Position end
 local o,v=pcall(function() return f:GetPivot().Position end)
 return o and v or nil
end
local function hp(f)
 local h=ch[f.Name]
 if type(h)~="number" then
  local s=sc and sc[f.Name]
  h=(type(s)=="table" and tonumber(s.h)) or tonumber(f:GetAttribute("Health")) or 0
 end
 return h
end
local function cok(f)
 if not(f and f.Parent) then return false end
 if f:GetAttribute("ID")==nil then return false end
 if sc and sc[f.Name]==nil then return false end
 if (blk[f.Name] or 0)>os.clock() then return false end
 local a=f:GetAttribute("Area")
 if type(a)=="string" and a~="" then
  if not ha(a) then return false end
  local rl=L.RebornLock
  if rl and type(rl.CoinAreaOpen)=="function" then
   local o,r=gc(rl.CoinAreaOpen,a)
   if o and r==false then return false end
  end
 end
 local m=f:FindFirstChild("Coin")
 if m and m:GetAttribute("PreventClick") then return false end
 return hp(f)>0
end
local function rf()
 if os.clock()-sca<5 then return end
 sca=os.clock()
 local o,r=inv("Get Coins")
 if o and type(r)=="table" then
  sc=r
  for id,d in pairs(r) do
   if type(d)=="table" and type(d.h)=="number" and ch[id]==nil then ch[id]=d.h end
  end
 end
end
local function rel(u,id)
 if id then inv("Leave Coin",id,u) end
 for _,x in ipairs(u) do fire("Change Pet Target",x,"Player") end
end
local function jn(f)
 local u=pets()
 if #u==0 then return false end
 local id=f:GetAttribute("ID") or f.Name
 local o,r=inv("Join Coin",id,u)
 if not o or type(r)~="table" then return false end
 local j={}
 for _,x in ipairs(u) do
  if r[x]==true then
   j[#j+1]=x
   fire("Change Pet Target",x,"Coin",id)
   fire("Farm Coin",id,x)
  end
 end
 if #j==0 then
  blk[id]=os.clock()+30
  rel(u,lid)
  return false
 end
 tg,tp,jt,lh,lid=f,j,os.clock(),0,id
 return true
end
local function ich(c)
 local a=c:GetAttribute("Name")
 return type(a)=="string" and a:find("Grand Heaven Chest",1,true)~=nil
end
local function ot()
 if not st.coin then return end
 local f=tf("Orbs")
 if not f then return end
 local n,b=os.clock(),{}
 for _,p in ipairs(f:GetChildren()) do
  local i=p.Name
  local e=ob[i]
  if not e then e={0,0} ob[i]=e end
  if n-e[1]>4 then
   if e[2]>=3 then
    pcall(function() p:Destroy() end)
    ob[i]=nil
   else
    e[1]=n
    e[2]+=1
    b[#b+1]=i
   end
  end
 end
 if #b>0 then fire("Claim Orbs",b) end
 for i in pairs(ob) do
  if not f:FindFirstChild(i) then ob[i]=nil end
 end
end
local function lo()
 if not st.loot then return end
 local f=tf("Lootbags")
 if not f then return end
 local n=os.clock()
 for _,p in ipairs(f:GetChildren()) do
  if p:IsA("BasePart") then
   local i=p:GetAttribute("ID")
   if i and not p:GetAttribute("Collected") and (not lt[i] or n-lt[i]>3) then
    lt[i]=n
    fire("Collect Lootbag",i,p.Position)
   end
  end
 end
 for i,t in pairs(lt) do
  if n-t>30 then lt[i]=nil end
 end
end

local function fc()
 local f=tf("Coins")
 local r=hr()
 if not(f and r) then return end
 local n=os.clock()
 if not ccl or #ccl==0 or n-cnx>0.5 then
  cnx=n
  ccl={}
  for _,c in ipairs(f:GetChildren()) do
   if ich(c) then ccl[#ccl+1]=c end
  end
 end
 local best,bd
 for _,c in ipairs(ccl) do
  if c.Parent then
   local p=cp(c)
   local d=p and (p-r.Position).Magnitude or math.huge
   if p and d<=math.huge and (not bd or d<bd) then best,bd=c,d end
  end
 end
 if best then
  local m=best:FindFirstChild("Coin")
  return best,m and m:FindFirstChildOfClass("ClickDetector"),cp(best),bd
 end
end
local function ck(cd,p)
 local k=MN[st.mt]
 if (k=="Fire" or k=="All") and cd and type(fireclickdetector)=="function" then
  pcall(fireclickdetector,cd)
 end
 if (k=="Connections" or k=="All") and cd and type(getconnections)=="function" then
  local o,cs=pcall(getconnections,cd.MouseClick)
  if o and type(cs)=="table" then
   for _,c in ipairs(cs) do pcall(function() c:Fire(lp) end) end
  end
 end
 if (k=="Mouse" or k=="All") and vim and p then
  local v,on=workspace.CurrentCamera:WorldToViewportPoint(p)
  if on then
   pcall(function()
    vim:SendMouseButtonEvent(v.X,v.Y,0,true,game,0)
    vim:SendMouseButtonEvent(v.X,v.Y,0,false,game,0)
   end)
  end
 end
end
local function cpt(c)
 if ctg and not tg then ctg=false end
 if ctg then
  if not cok(tg) then
   tg=nil
   ctg=false
  elseif os.clock()-math.max(jt,lh)>20 then
   rel(tp,tg:GetAttribute("ID") or tg.Name)
   tg=nil
   ctg=false
  end
  if ctg then return end
 end
 local n=os.clock()
 if n-cjr<0.25 or not cda() then return end
 if sc and sc[c.Name]==nil then sca=0 end
 rf()
 if not cok(c) then return end
 cjr=n
 if tg then
  rel(tp,tg:GetAttribute("ID") or tg.Name)
  tg=nil
 end
 if jn(c) then ctg=true end
end
local lbl
local function ct()
 local c,cd,p,d=fc()
 local t=c and string.format("Chest: %s (%d studs)",c.Name,d) or "Chest: none"
 if lbl and lbl.Text~=t then lbl.Text=t end
 local k=MN[st.mt]
 local pm=st.ck and (k=="Pets" or k=="All")
 if st.ck and c then ck(cd,p) end
 if c and pm then
  cpt(c)
 elseif ctg and tg then
  rel(tp,tg:GetAttribute("ID") or tg.Name)
  tg=nil
  ctg=false
 end
end

 on("Update Coin Health",function(id,h)

  end

  if type(id)~="string" then return end

   for _,x in ipairs(tp) do fire("Change Pet Target",x,"Player") end
   tg,tp=nil,{}
   ctg=false
  end
 end)
 on("Orb Removed",function(id) ob[id]=nil end)
 on("Remove Lootbag",function(id)
  if type(id)~="string" then return end
  lt[id]=nil
  local f=tf("Lootbags")
  local p=f and f:FindFirstChild(id)
  if p then pcall(function() p:Destroy() end) end
 end)
end)

local function dr(h,t,ck)
 local d,ds,sp,mv=false,nil,nil,false
 cn[#cn+1]=h.InputBegan:Connect(function(i)
  if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
   d,mv,ds,sp=true,false,i.Position,t.Position
   local c
   c=i.Changed:Connect(function()
    if i.UserInputState==Enum.UserInputState.End then
     d=false
     c:Disconnect()
     if ck and not mv then ck() end
    end
   end)
  end
 end)
 cn[#cn+1]=U.InputChanged:Connect(function(i)
  if d and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
   local v=i.Position-ds
   if v.Magnitude>5 then mv=true end
   if mv then t.Position=UDim2.new(sp.X.Scale,sp.X.Offset+v.X,sp.Y.Scale,sp.Y.Offset+v.Y) end
  end
 end)
end
dr(tb,fr)
